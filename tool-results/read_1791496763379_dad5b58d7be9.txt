    41→async function callLLM(userPrompt: string, usage: UsageAcc): Promise<string> {
    42→  const zai = await ZAI.create();
    43→  const t0 = Date.now();
    44→  const completion = await zai.chat.completions.create({
    45→    messages: [
    46→      { role: "system", content: SYSTEM_BASE },
    47→      { role: "user", content: userPrompt },
    48→    ],
    49→  });
    50→  const latencyMs = Date.now() - t0;
    51→  usage.calls += 1;
    52→  usage.latencyMs += latencyMs;
    53→  // chars/4 heuristic over prompt + completion (I13 estimate)
    54→  const est = Math.ceil((userPrompt.length + (completion.choices[0]?.message?.content?.length ?? 0)) / 4);
    55→  usage.estTokens += est;
    56→  return completion.choices[0]?.message?.content ?? "";
    57→}

    59→function parseJson<T>(raw: string): T {
    60→  let text = raw.trim();
    61→  text = text.replace(/^```(?:json)?\s*/i, "").replace(/\s*```$/, "");
    62→  const start = text.search(/[{[]/);
    63→  if (start > 0) text = text.slice(start);
    64→  try {
    65→    return JSON.parse(text) as T;
    66→  } catch {
    67→    /* fall through */
    68→  }
    69→  const lastBrace = Math.max(text.lastIndexOf("}"), text.lastIndexOf("]"));
    70→  if (lastBrace > 0) {
    71→    try {
    72→      return JSON.parse(text.slice(0, lastBrace + 1)) as T;
    73→    } catch {
    74→      /* fall through to structural repair */
    75→    }
    76→  }
    77→  // Production hardening: long LLM outputs can be truncated mid-element.
    78→  // Rebuild a valid prefix: cut at the last structural boundary outside
    79→  // strings, strip dangling commas, then close all open brackets/strings.
    80→  const repaired = repairTruncatedJson(text);
    81→  if (repaired !== null) {
    82→    try {
    83→      return JSON.parse(repaired) as T;
    84→    } catch {
    85→      /* fall through */
    86→    }
    87→  }
    88→  throw new Error("Agent returned unparseable output");
    89→}
    90→
    91→function repairTruncatedJson(text: string): string | null {
    92→  // find last structural boundary outside strings
    93→  let inStr = false;
    94→  let esc = false;
    95→  let lastSafe = -1;
    96→  for (let i = 0; i < text.length; i++) {
    97→    const ch = text[i];
    98→    if (inStr) {
    99→      if (esc) esc = false;
   100→      else if (ch === "\\") esc = true;
   101→      else if (ch === '"') inStr = false;
   102→      continue;
   103→    }
   104→    if (ch === '"') {
   105→      inStr = true;
   106→      continue;
   107→    }
   108→    if (ch === "," || ch === "}" || ch === "]") lastSafe = i;
   109→  }
   110→  if (lastSafe < 0) return null;
   111→  let cut = text.slice(0, lastSafe + 1);
   112→  // re-scan the cut for unclosed structures
   113→  const stack: string[] = [];
   114→  let s = false;
   115→  let e = false;
   116→  for (const ch of cut) {
   117→    if (s) {
   118→      if (e) e = false;
   119→      else if (ch === "\\") e = true;
   120→      else if (ch === '"') s = false;
   121→      continue;
   122→    }
   123→    if (ch === '"') {
   124→      s = true;
   125→      continue;
   126→    }
   127→    if (ch === "{") stack.push("}");
   128→    else if (ch === "[") stack.push("]");
   129→    else if (ch === "}" || ch === "]") stack.pop();
   130→  }
   131→  if (s) cut += '"';
   132→  cut = cut.replace(/,\s*$/, "");
   133→  while (stack.length) cut += stack.pop();
   134→  return cut;
   135→}

   146→        estCostUsd: +(u.estTokens * 0.0000006).toFixed(6), // blended $0.60 / 1M tokens (estimate)
   147→        promptVersion: PROMPT_VERSION,

   151→    /* ignore */

   157→function evidenceBlock(evidences: { code: string; title: string; sourceType: string; language: string; content: string }[]): string {
   158→  if (!evidences.length) return "(no evidence sources provided yet)";
   159→  return evidences
   160→    .map(
   161→      (e) =>
   162→        `[${e.code}] (${e.sourceType}, language: ${e.language}) "${e.title}"\n${e.content.slice(0, 1200)}`
   163→    )
   164→    .join("\n\n");
   165→}

   171→async function memoryBlock(): Promise<string> {
   172→  const lessons = await db.memoryEntry.findMany({
   173→    where: { kind: "lesson" },

   179→    .map((l) => `- ${l.term}: ${l.detail}`)

   183→function uncertaintyBlock(questions: { text: string; fallbackAssumption: string | null; resolution: string }[]): string {
   184→  const decided = questions.filter((q) => q.resolution === "proceed_with_uncertainty" && q.fallbackAssumption);
   185→  if (!decided.length) return "";
   186→  return `\nUNCERTAINTY LEDGER (the human chose to proceed WITHOUT asking these — treat the stated assumption as an explicit, auditable assumption):\n${decided
   187→    .map((q) => `- Assume: ${q.fallbackAssumption} (instead of asking: "${q.text.slice(0, 120)}")`)
   188→    .join("\n")}\n`;
   189→}

   320→export async function runExtraction(projectId: string, runId?: string): Promise<void> {
   321→  const project = await db.project.findUniqueOrThrow({
   322→    where: { id: projectId },
   323→    include: { evidences: true, questions: true, candidates: true },
   324→  });
   325→  const usage = newUsage();
   326→
   327→  const answers = project.questions
   328→    .filter((q) => q.answered && q.answer)
   329→    .map((q) => `Q: ${q.text}\nA: ${q.answer}`)
   330→    .join("\n\n");
   331→
   332→  const selected = project.candidates.find((c) => c.selected);
   333→  const evidences = project.evidences;
   334→  const codeToId = new Map(evidences.map((e) => [e.code, e.id]));
   335→
   336→  const prompt = `PROJECT BRIEF:
   337→${briefBlock(project.brief)}
   338→
   339→SELECTED PROBLEM STATEMENT:
   340→${selected ? `${selected.title} — ${selected.description}` : "(none selected — cover the whole brief)"}
   341→
   342→EVIDENCE SOURCES:
   343→${evidenceBlock(evidences)}
   344→
   345→STAKEHOLDER ANSWERS:
   346→${answers || "(no answers yet)"}
   347→${uncertaintyBlock(project.questions)}
   348→TASK: Extract 8-12 atomic requirements for the selected problem. Each requirement must be a single testable statement ("The system shall …"). Cite the evidence codes that ground it. Set priority by MoSCoW. Keep the original language of the source when a requirement is grounded in Arabic or code-switched evidence (set language accordingly: "en", "ar-EG" or "codeswitch"); otherwise use English. Also report your confidence (0-100) that each requirement is correct, complete and correctly grounded.
   349→
   350→Return JSON: {"requirements": [{"text": "...", "type": "functional|nonfunctional|constraint", "priority": "Must|Should|Could|Won't", "evidenceCodes": ["E1"], "confidence": 82, "language": "en"}]}`;
   351→
   352→  const out = parseJson<{ requirements: AgentRequirement[] }>(await callLLM(prompt, usage));
   353→  const rs = (out.requirements ?? []).slice(0, 14);
   354→
   355→  // replace previous extraction
   356→  await db.requirement.deleteMany({ where: { projectId } });
   357→  let i = 0;
   358→  const refIndex: { refId: string; id: string; text: string }[] = [];
   359→  for (const r of rs) {
   360→    i += 1;
   361→    const refId = `REQ-${String(i).padStart(2, "0")}`;
   362→    const type = ["functional", "nonfunctional", "constraint"].includes(r.type)
   363→      ? r.type
   364→      : "functional";
   365→    const priority = ["Must", "Should", "Could", "Won't"].includes(r.priority)
   366→      ? r.priority
   367→      : "Should";
   368→    const codes = (r.evidenceCodes ?? []).filter((c) => codeToId.has(c));
   369→    const created = await db.requirement.create({
   370→      data: {
   371→        projectId,
   372→        refId,
   373→        text: String(r.text ?? "").slice(0, 800),
   374→        type,
   375→        priority,
   376→        language: ["en", "ar-EG", "codeswitch"].includes(r.language) ? r.language : "en",
   377→        confidence: Math.min(100, Math.max(0, Math.round(Number(r.confidence) || 60))),
   378→        ambiguity: -1,
   379→        order: i,
   380→      },
   381→    });
   382→    refIndex.push({ refId, id: created.id, text: created.text });
   383→    for (const c of codes) {
   384→      await db.traceLink.create({
   385→        data: { projectId, requirementId: created.id, evidenceCode: c, note: "cited by extraction agent" },
   386→      });
   387→    }
   388→  }
   389→
   390→  // ── I3 critic-verifier pass: second model call checks each
   391→  //    requirement against the evidence spans (Propose → Verify).
   392→  try {
   393→    const verifyPrompt = `You are the VERIFIER in a propose-verify requirements pipeline. Below are candidate requirements and the evidence sources they claim to be grounded in. For each requirement, judge strictly whether the evidence actually supports it:
   394→- "supported": the evidence explicitly states or clearly implies it.
   395→- "weak": the evidence only partially supports it, is stretched, or the requirement adds an assumption not in the evidence.
   396→
   397→EVIDENCE SOURCES:
   398→${evidenceBlock(evidences)}
   399→
   400→CANDIDATE REQUIREMENTS:
   401→${refIndex.map((r) => `${r.refId}: ${r.text}`).join("\n")}
   402→
   403→Return JSON: {"verdicts": [{"refId": "REQ-01", "verdict": "supported|weak", "note": "one-line justification"}]}`;
   404→
   405→    const vOut = parseJson<{ verdicts: Verdict[] }>(await callLLM(verifyPrompt, usage));
   406→    for (const v of vOut.verdicts ?? []) {
   407→      const target = refIndex.find((r) => r.refId === v.refId);
   408→      if (!target) continue;
   409→      await db.requirement.update({
   410→        where: { id: target.id },
   411→        data: {
   412→          verifyStatus: v.verdict === "supported" ? "supported" : "weak",
   413→          verifyNote: String(v.note ?? "").slice(0, 400),
   414→        },
   415→      });
   416→    }
   417→  } catch {
   418→    // verifier is best-effort; requirements stay "unverified" on failure
   419→  }
   420→
   421→  await saveArtifact(projectId, "EXTRACTION", out);
   422→  await recordUsage(runId, usage);
   423→}

   527→// ---------- stage 6: CONFLICTS ----------
   528→
   529→export interface AgentConflict {
   530→  reqARef: string;
   531→  reqBRef: string;
   532→  kind: "contradiction" | "duplication" | "overlap";
   533→  description: string;
   534→  severity: "high" | "medium" | "low";
   535→  suggestion: string;
   536→}
   537→
   538→export async function runConflicts(projectId: string, runId?: string): Promise<void> {
   539→  const project = await db.project.findUniqueOrThrow({
   540→    where: { id: projectId },
   541→    include: { reqs: { orderBy: { order: "asc" } } },
   542→  });
   543→  const usage = newUsage();
   544→
   545→  const list = project.reqs.map((r) => `${r.refId}: ${r.text}`).join("\n");
   546→
   547→  const prompt = `REQUIREMENTS:
   548→${list}
   549→
   550→TASK: Cross-check all requirement pairs for (a) contradictions — cannot both hold, (b) duplications — same need stated twice, (c) overlaps — partially conflicting scopes. Only report genuine findings (0 to 5). For each: the two refIds, kind, a 1-2 sentence explanation, severity, and a suggested resolution in the requirements' own language.
   551→
   552→Return JSON: {"conflicts": [{"reqARef": "REQ-01", "reqBRef": "REQ-04", "kind": "contradiction", "description": "...", "severity": "high", "suggestion": "..."}]}`;
   553→
   554→  const out = parseJson<{ conflicts: AgentConflict[] }>(await callLLM(prompt, usage));
   555→  const byRef = new Map(project.reqs.map((r) => [r.refId, r]));
   556→
   557→  await db.conflict.deleteMany({ where: { projectId } });
   558→  for (const c of (out.conflicts ?? []).slice(0, 6)) {
   559→    const a = byRef.get(c.reqARef);
   560→    const b = byRef.get(c.reqBRef);
   561→    if (!a || !b || a.id === b.id) continue;
   562→    await db.conflict.create({
   563→      data: {
   564→        projectId,
   565→        reqAId: a.id,
   566→        reqBId: b.id,
   567→        kind: ["contradiction", "duplication", "overlap"].includes(c.kind) ? c.kind : "overlap",
   568→        description: String(c.description ?? "").slice(0, 800),
   569→        severity: ["high", "medium", "low"].includes(c.severity) ? c.severity : "medium",
   570→        suggestion: String(c.suggestion ?? "").slice(0, 800),
   571→      },
   572→    });
   573→  }
   574→  await saveArtifact(projectId, "CONFLICTS", out);
   575→  await recordUsage(runId, usage);
   576→}

   578→// ---------- I7 synthetic stakeholder wind tunnel ----------
   579→
   580→export async function runWindTunnel(
   581→  projectId: string,
   582→  persona: string,
   583→  secret: string
   584→): Promise<{ answered: number }> {
   585→  const project = await db.project.findUniqueOrThrow({
   586→    where: { id: projectId },
   587→    include: { questions: true },
   588→  });
   589→  const usage = newUsage();
   590→
   591→  const pending = project.questions.filter((q) => !q.answered);
   592→  if (!pending.length) return { answered: 0 };
   593→
   594→  const prompt = `You are simulating a SYNTHETIC STAKEHOLDER in a requirements "wind tunnel". The purpose is stress-testing the clarifying questions before a real interview.
   595→
   596→PERSONA: ${persona}
   597→
   598→HELD-OUT SECRET (the persona knows this but will only reveal it if a question DIRECTLY probes the topic; otherwise give an evasive, vague or partially relevant answer): ${secret || "(none)"}
   599→
   600→QUESTIONS:
   601→${pending.map((q) => `[${q.id}] ${q.text}`).join("\n")}
   602→
   603→TASK: Answer each question in character. If (and only if) the question directly probes the secret's topic, reveal it naturally. Otherwise answer plausibly but incompletely. Answers should be in the persona's voice; mirror the language style of the question (English, Egyptian Arabic or code-switched).
   604→
   605→Return JSON: {"answers": [{"id": "question id", "answer": "..."}]}`;
   606→
   607→  const out = parseJson<{ answers: { id: string; answer: string }[] }>(await callLLM(prompt, usage));
   608→  let n = 0;
   609→  for (const a of out.answers ?? []) {
   610→    const q = pending.find((p) => p.id === a.id);
   611→    if (!q || !a.answer?.trim()) continue;
   612→    await db.question.update({
   613→      where: { id: q.id },
   614→      data: {
   615→        answer: `[synthetic · wind tunnel] ${String(a.answer).slice(0, 1500)}`,
   616→        answered: true,
   617→        resolution: "answered",
   618→      },
   619→    });
   620→    n += 1;
   621→  }
   622→  await logDecision(
   623→    projectId,
   624→    "ELICITATION",
   625→    "agent",
   626→    "synthetic_answers",
   627→    `Wind tunnel (${persona.slice(0, 80)}) filled ${n} answer(s) with persona-conditioned synthetic responses.`
   628→  );
   629→  await saveArtifact(projectId, "WIND_TUNNEL", { persona, secret, answers: out.answers ?? [] });
   630→  return { answered: n };
   631→}

   635→export async function runDrift(projectId: string, signal: string): Promise<{ impacted: number; summary: string }> {
   636→  const project = await db.project.findUniqueOrThrow({
   637→    where: { id: projectId },
   638→    include: { reqs: { orderBy: { order: "asc" } } },
   639→  });
   640→  const usage = newUsage();
   641→
   642→  const prompt = `REQUIREMENTS BASELINE:
   643→${project.reqs.map((r) => `${r.refId}: ${r.text}`).join("\n")}
   644→
   645→CHANGE SIGNAL (something that changed in the project's world after the baseline was frozen):
   646→${signal}
   647→
   648→TASK: Determine which requirements are impacted by this change signal (contradicted, invalidated, now incomplete, or needing revision). Be conservative — only flag genuinely impacted requirements (0 to all). Give a one-paragraph summary.
   649→
   650→Return JSON: {"impacted": [{"refId": "REQ-03", "note": "why it is impacted"}], "summary": "..."}`;
   651→
   652→  const out = parseJson<{ impacted: { refId: string; note: string }[]; summary: string }>(await callLLM(prompt, usage));
   653→
   654→  await db.requirement.updateMany({ where: { projectId }, data: { driftStatus: "none" } });
   655→  for (const imp of out.impacted ?? []) {
   656→    const r = project.reqs.find((x) => x.refId === imp.refId);
   657→    if (!r) continue;
   658→    await db.requirement.update({
   659→      where: { id: r.id },
   660→      data: { driftStatus: "suspected" },
   661→    });
   662→  }
   663→  await logDecision(projectId, "LIFECYCLE", "agent", "drift_check", out.summary ?? "");
   664→  await saveArtifact(projectId, "DRIFT", out);
   665→  return { impacted: (out.impacted ?? []).length, summary: String(out.summary ?? "") };
   666→}

   720→// ---------- I14 requirements health report ----------
   721→
   722→export interface HealthReport {
   723→  summary: string;
   724→  strengths: string[];
   725→  risks: { title: string; note: string }[];
   726→  recommendations: string[];
   727→  ambiguityDebt: string;
   728→  uncertaintyLedger: string;
   729→  conflictDigest: string;
   730→}
   731→
   732→export async function runHealth(projectId: string): Promise<HealthReport> {
   733→  const project = await db.project.findUniqueOrThrow({
   734→    where: { id: projectId },
   735→    include: { reqs: true, conflicts: { include: { reqA: true, reqB: true } }, questions: true, candidates: true, stories: true },
   736→  });
   737→  const usage = newUsage();
   738→
   739→  const assessed = project.reqs.filter((r) => r.ambiguity >= 0);
   740→  const avgAmb = assessed.length ? Math.round(assessed.reduce((s, r) => s + r.ambiguity, 0) / assessed.length) : -1;
   741→  const flagged = assessed.filter((r) => r.ambiguity >= 30);
   742→  const uncertain = project.questions.filter((q) => q.resolution === "proceed_with_uncertainty");
   743→  const unresolvedConflicts = project.conflicts.filter((c) => c.status === "open");
   744→  const selected = project.candidates.find((c) => c.selected);
   745→
   746→  const stats = `PROJECT: ${project.name}
   747→SELECTED PROBLEM: ${selected ? selected.title : "(none)"}
   748→REQUIREMENTS: ${project.reqs.length} total; avg ambiguity ${avgAmb < 0 ? "n/a" : avgAmb}; ${flagged.length} flagged >= 30; weak-verification: ${project.reqs.filter((r) => r.verifyStatus === "weak").length}; low-confidence (<50): ${project.reqs.filter((r) => r.confidence >= 0 && r.confidence < 50).length}
   749→UNCERTAINTY LEDGER: ${uncertain.length} assumption(s) carried without asking:
   750→${uncertain.map((q) => `- ${q.fallbackAssumption ?? q.text}`).join("\n") || "(none)"}
   751→CONFLICTS: ${project.conflicts.length} detected, ${unresolvedConflicts.length} still open
   752→STORIES: ${project.stories.length}`;
   753→
   754→  const prompt = `${stats}
   755→
   756→TASK: Produce a requirements health report for handoff. Write it for a delivery team lead. Be specific and honest; do not pad. ambiguityDebt / uncertaintyLedger / conflictDigest must each be 1-2 sentences quantifying that debt area.
   757→
   758→Return JSON: {"summary": "2-3 sentence overall health assessment", "strengths": ["..."], "risks": [{"title": "...", "note": "..."}], "recommendations": ["..."], "ambiguityDebt": "...", "uncertaintyLedger": "...", "conflictDigest": "..."}`;
   759→
   760→  const out = parseJson<HealthReport>(await callLLM(prompt, usage));
   761→  await logDecision(projectId, "LIFECYCLE", "agent", "health_report", out.summary ?? "");
   762→  await saveArtifact(projectId, "HEALTH", out);
   763→  return out;
   764→}
