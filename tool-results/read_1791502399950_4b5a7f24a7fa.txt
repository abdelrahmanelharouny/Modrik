    74→function download(name: string, content: string, type = "application/json") {
    75→  const blob = new Blob([content], { type });
    76→  const url = URL.createObjectURL(blob);
    77→  const a = document.createElement("a");
    78→  a.href = url;
    79→  a.download = name;
    80→  a.click();
    81→  URL.revokeObjectURL(url);
    82→}

    85→function Constellation({ project }: { project: ProjectDTO }) {
    86→  const [sel, setSel] = useState<{ kind: "req" | "ev"; id: string } | null>(null);
    87→  const W = 640;
    88→  const H = 400;
    89→  const cx = W / 2;
    90→  const cy = H / 2;
    91→  const reqR = 118;
    92→  const evR = 176;
    93→
    94→  const reqs = project.requirements;
    95→  const evs = project.evidences;
    96→  const reqPos = useMemo(() => {
    97→    const m = new Map<string, { x: number; y: number }>();
    98→    reqs.forEach((r, i) => {
    99→      const a = (i / Math.max(1, reqs.length)) * 2 * Math.PI - Math.PI / 2;
   100→      m.set(r.id, { x: cx + reqR * Math.cos(a), y: cy + reqR * Math.sin(a) });
   101→    });
   102→    return m;
   103→  }, [reqs, cx, cy]);
   104→  const evPos = useMemo(() => {
   105→    const m = new Map<string, { x: number; y: number }>();
   106→    evs.forEach((e, i) => {
   107→      const a = (i / Math.max(1, evs.length)) * 2 * Math.PI - Math.PI / 2 + Math.PI / Math.max(1, evs.length);
   108→      m.set(e.code, { x: cx + evR * Math.cos(a), y: cy + evR * Math.sin(a) });
   109→    });
   110→    return m;
   111→  }, [evs, cx, cy]);
   112→
   113→  const selectedReq = sel?.kind === "req" ? reqs.find((r) => r.id === sel.id) : null;
   114→  const selectedEv = sel?.kind === "ev" ? evs.find((e) => e.id === sel.id) : null;
   115→
   116→  if (reqs.length === 0) {
   117→    return <EmptyBay>The constellation lights up once requirements exist — approve the Airframe phase first.</EmptyBay>;
   118→  }
   119→
   120→  return (
   121→    <div className="space-y-3">
   122→      <div className="overflow-x-auto rounded-md border border-border/70 bg-abyss/60">
   123→        <svg viewBox={`0 0 ${W} ${H}`} className="mx-auto block h-[340px] w-full min-w-[520px] sm:h-[400px]" role="img" aria-label="Requirement constellation">
   124→          {/* evidence grounding edges */}
   125→          {reqs.map((r) =>
   126→            r.links.map((l) => {
   127→              const p1 = reqPos.get(r.id);
   128→              const p2 = evPos.get(l.evidenceCode);
   129→              if (!p1 || !p2) return null;
   130→              const active = sel?.kind === "req" ? sel.id === r.id : sel?.kind === "ev" ? sel.id === evs.find((e) => e.code === l.evidenceCode)?.id : true;
   131→              return (
   132→                <line key={`${r.id}-${l.evidenceCode}`} x1={p1.x} y1={p1.y} x2={p2.x} y2={p2.y} stroke="rgba(44,232,200,0.35)" strokeWidth={active ? 1.4 : 0.5} opacity={active ? 0.9 : 0.25}>
   133→                  <animate attributeName="stroke-dasharray" values="1 7;3 7;1 7" dur="3s" repeatCount="indefinite" />
   134→                </line>
   135→              );
   136→            })
   137→          )}
   138→          {/* conflict edges */}
   139→          {project.conflicts.map((c) => {
   140→            const p1 = reqPos.get(c.reqAId);
   141→            const p2 = reqPos.get(c.reqBId);
   142→            if (!p1 || !p2) return null;
   143→            const resolved = c.status === "resolved";
   144→            return (
   145→              <line key={c.id} x1={p1.x} y1={p1.y} x2={p2.x} y2={p2.y} stroke={resolved ? "rgba(61,220,151,0.5)" : "rgba(255,107,122,0.65)"} strokeWidth="1.2" strokeDasharray="4 4">
   146→                <animate attributeName="stroke-dashoffset" values="0;-16" dur="1.6s" repeatCount="indefinite" />
   147→              </line>
   148→            );
   149→          })}
   150→          {/* evidence nodes */}
   151→          {evs.map((e) => {
   152→            const p = evPos.get(e.code);
   153→            if (!p) return null;
   154→            const active = sel?.id === e.id;
   155→            return (
   156→              <g key={e.id} onClick={() => setSel(active ? null : { kind: "ev", id: e.id })} className="cursor-pointer">
   157→                <circle cx={p.x} cy={p.y} r="13" fill="rgba(167,139,250,0.12)" stroke="rgba(167,139,250,0.5)" strokeWidth={active ? 2 : 1} />
   158→                <text x={p.x} y={p.y + 3} textAnchor="middle" fill="#a78bfa" fontSize="9" fontFamily="var(--font-jbmono), monospace">{e.code}</text>
   159→              </g>
   160→            );
   161→          })}
   162→          {/* requirement nodes */}
   163→          {reqs.map((r) => {
   164→            const p = reqPos.get(r.id);
   165→            if (!p) return null;
   166→            const active = sel?.id === r.id;
   167→            const flagged = r.driftStatus !== "none";
   168→            const color = flagged ? "#ffb454" : r.ambiguity >= 60 ? "#ff6b7a" : r.ambiguity >= 30 ? "#ffb454" : "#2ce8c8";
   169→            return (
   170→              <g key={r.id} onClick={() => setSel(active ? null : { kind: "req", id: r.id })} className="cursor-pointer">
   171→                <circle cx={p.x} cy={p.y} r={active ? 17 : 14} fill={active ? "rgba(44,232,200,0.18)" : "rgba(44,232,200,0.07)"} stroke={color} strokeWidth={active ? 2 : 1.2} />
   172→                <circle cx={p.x} cy={p.y} r="3" fill={color}>
   173→                  <animate attributeName="opacity" values="1;0.35;1" dur="2.4s" repeatCount="indefinite" />
   174→                </circle>
   175→                <text x={p.x} y={p.y + 26} textAnchor="middle" fill="rgba(230,238,245,0.85)" fontSize="8.5" fontFamily="var(--font-jbmono), monospace">{r.refId}</text>
   176→              </g>
   177→            );
   178→          })}
   179→          {/* hub */}
   180→          <circle cx={cx} cy={cy} r="4" fill="#2ce8c8" opacity="0.9" />
   181→          <circle cx={cx} cy={cy} r="12" fill="none" stroke="rgba(44,232,200,0.3)" strokeWidth="1">
   182→            <animate attributeName="r" values="10;22;10" dur="3s" repeatCount="indefinite" />
   183→            <animate attributeName="opacity" values="0.7;0;0.7" dur="3s" repeatCount="indefinite" />
   184→          </circle>
   185→        </svg>
   186→      </div>
   187→      <div className="flex flex-wrap items-center gap-3 font-mono text-[10px] uppercase tracking-wider text-muted-foreground">
   188→        <span className="flex items-center gap-1.5"><span className="h-2 w-2 rounded-full bg-aura" /> requirement (colour = ambiguity / drift)</span>
   189→        <span className="flex items-center gap-1.5"><span className="h-2 w-2 rounded-full bg-plasma" /> evidence</span>
   190→        <span className="flex items-center gap-1.5"><span className="h-0.5 w-5 bg-coral" /> open conflict</span>
   191→        <span className="flex items-center gap-1.5"><span className="h-0.5 w-5 bg-verify" /> resolved</span>
   192→        <span className="ml-auto text-muted-foreground/60">click a node</span>
   193→      </div>
   194→      {selectedReq && (
   195→        <HudPanel className="p-4" as="div">
   196→          <div className="mb-1 flex items-center gap-2">
   197→            <Badge variant="outline" className="border-aura/40 bg-aura/5 font-mono text-xs text-aura">{selectedReq.refId}</Badge>
   198→            <Chip tone={selectedReq.priority === "Must" ? "coral" : selectedReq.priority === "Should" ? "signal" : "aura"}>{selectedReq.priority}</Chip>
   199→            <Chip tone="muted">amb {selectedReq.ambiguity < 0 ? "—" : selectedReq.ambiguity}</Chip>
   200→            {selectedReq.driftStatus !== "none" && <Chip tone="signal">drift: {selectedReq.driftStatus}</Chip>}
   201→          </div>
   202→          <p className="text-sm" dir="auto">{selectedReq.text}</p>
   203→        </HudPanel>
   204→      )}
   205→      {selectedEv && (
   206→        <HudPanel className="p-4" as="div">
   207→          <div className="mb-1 flex items-center gap-2">
   208→            <Badge variant="outline" className="border-plasma/40 bg-plasma/5 font-mono text-xs text-plasma">{selectedEv.code}</Badge>
   209→            <span className="text-sm font-medium" dir="auto">{selectedEv.title}</span>
   210→          </div>
   211→          <p className="line-clamp-4 text-sm text-muted-foreground" dir="auto">{selectedEv.content}</p>
   212→        </HudPanel>
   213→      )}
   214→    </div>
   215→  );
   216→}

   218→// ── main bay ─────────────────────────────────────────────────
   219→export function DebriefBay({ project, refresh }: { project: ProjectDTO; refresh: () => Promise<void> }) {
   220→  const { toast } = useToast();
   221→  const { lang } = useLang();
   222→  const ar = lang === "ar";
   223→
   224→  const [health, setHealth] = useState<HealthReport | null>(null);
   225→  const [healthBusy, setHealthBusy] = useState(false);
   226→  const [drift, setDrift] = useState<DriftResult | null>(null);
   227→  const [driftBusy, setDriftBusy] = useState(false);
   228→  const [signal, setSignal] = useState("");
   229→  const [compliance, setCompliance] = useState<Record<string, ComplianceResult>>({});
   230→  const [packBusy, setPackBusy] = useState<string | null>(null);
   231→  const [persona, setPersona] = useState("");
   232→  const [secret, setSecret] = useState("");
   233→  const [tunnel, setTunnel] = useState<{ answered: number } | null>(null);
   234→  const [tunnelBusy, setTunnelBusy] = useState(false);
   235→  const [copied, setCopied] = useState<string | null>(null);
   236→
   237→  useEffect(() => {
   238→    api<{ report: HealthReport | null }>(`/api/projects/${project.id}/health`)
   239→      .then((d) => setHealth(d.report))
   240→      .catch(() => {});
   241→  }, [project.id]);
   242→
   243→  const runHealth = useCallback(async () => {
   244→    setHealthBusy(true);
   245→    try {
   246→      const d = await api<{ report: HealthReport }>(`/api/projects/${project.id}/health`, { method: "POST" });
   247→      setHealth(d.report);
   248→      toast({ title: "Health report generated", description: "Decomposed dimensions, no magic score." });
   249→    } catch (e) {
   250→      toast({ title: "Health run failed", description: (e as Error).message, variant: "destructive" });
   251→    } finally {
   252→      setHealthBusy(false);
   253→    }
   254→  }, [project.id, toast]);
   255→
   256→  const runDrift = async () => {
   257→    if (!signal.trim()) return;
   258→    setDriftBusy(true);
   259→    try {
   260→      const d = await api<DriftResult>(`/api/projects/${project.id}/drift`, { method: "POST", body: JSON.stringify({ signal }) });
   261→      setDrift(d);
   262→      await refresh();
   263→      toast({ title: "Drift sweep complete", description: `${d.impacted} requirement(s) flagged.` });
   264→    } catch (e) {
   265→      toast({ title: "Drift sweep failed", description: (e as Error).message, variant: "destructive" });
   266→    } finally {
   267→      setDriftBusy(false);
   268→    }
   269→  };
   270→
   271→  const clearDrift = async () => {
   272→    try {
   273→      await api(`/api/projects/${project.id}/drift`, { method: "DELETE" });
   274→      setDrift(null);
   275→      await refresh();
   276→    } catch (e) {
   277→      toast({ title: "Could not clear flags", description: (e as Error).message, variant: "destructive" });
   278→    }
   279→  };
   280→
   281→  const runPack = async (pack: string) => {
   282→    setPackBusy(pack);
   283→    try {
   284→      const d = await api<ComplianceResult>(`/api/projects/${project.id}/compliance`, { method: "POST", body: JSON.stringify({ pack }) });
   285→      setCompliance((c) => ({ ...c, [pack]: d }));
   286→      toast({ title: `${PACKS_UI.find((p) => p.id === pack)?.name} scanned`, description: `Overall: ${d.overall}` });
   287→    } catch (e) {
   288→      toast({ title: "Pack scan failed", description: (e as Error).message, variant: "destructive" });
   289→    } finally {
   290→      setPackBusy(null);
   291→    }
   292→  };
   293→
   294→  const runTunnel = async () => {
   295→    if (!persona.trim()) return;
   296→    setTunnelBusy(true);
   297→    try {
   298→      const d = await api<{ answered: number }>(`/api/projects/${project.id}/wind-tunnel`, {
   299→        method: "POST",
   300→        body: JSON.stringify({ persona, secret }),
   301→      });
   302→      setTunnel(d);
   303→      await refresh();
   304→      toast({ title: "Wind tunnel complete", description: `${d.answered} pending question(s) answered by the synthetic stakeholder.` });
   305→    } catch (e) {
   306→      toast({ title: "Wind tunnel failed", description: (e as Error).message, variant: "destructive" });
   307→    } finally {
   308→      setTunnelBusy(false);
   309→    }
   310→  };
   311→
   312→  const reopen = async () => {
   313→    try {
   314→      await api(`/api/projects/${project.id}/pipeline`, { method: "POST", body: JSON.stringify({ action: "reopen" }) });
   315→      await refresh();
   316→      toast({ title: "Pipeline reopened", description: "The baseline is unlocked for further iteration." });
   317→    } catch (e) {
   318→      toast({ title: "Could not reopen", description: (e as Error).message, variant: "destructive" });
   319→    }
   320→  };
   321→
   322→  // ── dynamic stats (all from project DTO) ──
   323→  const approved = project.requirements.filter((r) => r.priority !== "Won't");

   326→  const openConflicts = project.conflicts.filter((c) => c.status === "open").length;
   327→  const selected = project.candidates.find((c) => c.selected);
   328→  const flaggedReqs = project.requirements.filter((r) => r.driftStatus !== "none");
   329→  const assumptions = project.questions.filter((q) => q.resolution === "proceed_with_uncertainty");
   330→  const byPriority = PRIORITY_ORDER.map((p) => ({ p, n: approved.filter((r) => r.priority === p).length }));
   331→  const maxN = Math.max(1, ...byPriority.map((b) => b.n));
   332→
   333→  const stats = [
   334→    { icon: FileText, label: ar ? "مصادر الأدلة" : "Evidence sources", value: project.evidences.length },
   335→    { icon: ListChecks, label: ar ? "متطلبات في الخط" : "Requirements in baseline", value: approved.length },
   336→    { icon: Gauge, label: ar ? "متوسط الغموض" : "Avg ambiguity", value: avgAmb },
   337→    { icon: BookOpen, label: ar ? "قصص المستخدم" : "User stories", value: project.stories.length },
   338→    { icon: GitMerge, label: ar ? "تعارضات مفتوحة" : "Open conflicts", value: openConflicts },
   339→  ];
   340→
   341→  const exportJson = () => {
   342→    const payload = {

   348→      stories: project.stories,
   349→      conflicts: project.conflicts,
   350→    };
   351→    download(`${project.name.replace(/\W+/g, "_")}_requirements_baseline.json`, JSON.stringify(payload, null, 2));
   352→    toast({ title: "Baseline exported" });
   353→  };

   367→    toast({ title: "Gherkin skeleton exported" });
   368→  };
   369→
   370→  const exportAudit = () => {
   371→    download(
   372→      `${project.name.replace(/\W+/g, "_")}_audit_log.json`,
   373→      JSON.stringify({ project: project.name, decisions: project.decisions, runs: project.runs }, null, 2)
   374→    );
   375→    toast({ title: "Audit log exported" });
   376→  };
   377→
   378→  const githubPayload = () =>
   379→    JSON.stringify(
   380→      {
   381→        repository_target: "GitHub Issues (REST v3)",
   382→        mode: "draft payloads — OAuth write pass next",
   383→        issues: approved.map((r) => ({
   384→          title: `[${r.refId}] ${r.text.slice(0, 80)}`,
   385→          labels: ["requirement", `priority:${r.priority.toLowerCase()}`, `type:${r.type}`, `verify:${r.verifyStatus}`],
   386→          body: `${r.text}\n\nAmbiguity: ${r.ambiguity < 0 ? "n/a" : r.ambiguity}\nEvidence: ${r.links.map((l) => l.evidenceCode).join(", ") || "—"}`,
   387→        })),
   388→      },
   389→      null,
   390→      2
   391→    );
   392→  const jiraPayload = () =>
   393→    JSON.stringify(
   394→      {
   395→        target: "Jira Cloud REST API v3",
   396→        mode: "draft payloads — site auth pass next",
   397→        issues: project.stories.map((s) => ({
   398→          fields: {
   399→            summary: `${s.refId} ${s.title}`,
   400→            issuetype: { name: "Story" },
   401→            description: `${s.narrative}\n\nAcceptance criteria:\n${s.acceptance}\n\nTraceability: ${project.requirements.find((r) => r.id === s.requirementId)?.refId ?? "—"}`,
   402→            labels: ["requirementpilot", `invest:${s.investScore ?? "n/a"}`],
   403→          },
   404→        })),
   405→      },
   406→      null,
   407→      2
   408→    );

   410→  const copyPayload = (id: string, content: string) => {
   411→    navigator.clipboard.writeText(content).then(() => {
   412→      setCopied(id);
   413→      setTimeout(() => setCopied(null), 1600);
   414→    });

   421→    windtunnel: Wind,
   422→    constellation: Waypoints,
   423→    journal: ScrollText,
   424→  };

   426→  return (
   427→    <div className="space-y-6">
   428→      {/* landing strip */}
   429→      <Reveal>
   430→        <HudPanel glow className="p-6">
   431→          <div className="flex flex-wrap items-start justify-between gap-4">
   432→            <div className="flex items-center gap-3">
