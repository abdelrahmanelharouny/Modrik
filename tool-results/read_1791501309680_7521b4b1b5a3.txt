    41→    41→async function callLLM(userPrompt: string, usage: UsageAcc): Promise<string> {
    42→    42→  const zai = await ZAI.create();
    43→    43→  const t0 = Date.now();
    44→    44→  let completion: Awaited<ReturnType<typeof zai.chat.completions.create>>;
    45→    45→  try {
    46→    46→    completion = await zai.chat.completions.create({
    47→    47→      messages: [
    48→    48→        { role: "system", content: SYSTEM_BASE },
    49→    49→        { role: "user", content: userPrompt },
    50→    50→      ],
    51→    51→    });
    52→    52→  } catch (err) {
    53→    53→    // one transient-failure retry (network blips / 5xx) — the audit found a
    54→    54→    // single unguarded failure aborts a whole stage run
    55→    55→    await new Promise((r) => setTimeout(r, 1500));
    56→    56→    completion = await zai.chat.completions.create({
    57→    57→      messages: [
    58→    58→        { role: "system", content: SYSTEM_BASE },
    59→    59→        { role: "user", content: userPrompt },
    60→    60→      ],
    61→    61→    });
    62→    62→  }
    63→    63→  const latencyMs = Date.now() - t0;
    64→    64→  usage.calls += 1;
    65→    65→  usage.latencyMs += latencyMs;
    66→    66→  // chars/4 heuristic over prompt + completion (I13 estimate)
    67→    67→  const est = Math.ceil((userPrompt.length + (completion.choices[0]?.message?.content?.length ?? 0)) / 4);
    68→    68→  usage.estTokens += est;
    69→    69→  return completion.choices[0]?.message?.content ?? "";
    70→    70→}

    75→    75→async function callLLMJson<T>(userPrompt: string, usage: UsageAcc, attempts = 2): Promise<T> {
    76→    76→  let lastErr: unknown = new Error("Agent returned unparseable output");
    77→    77→  for (let i = 0; i < attempts; i++) {
    78→    78→    const raw = await callLLM(userPrompt, usage);
    79→    79→    try {
    80→    80→      return parseJson<T>(raw);
    81→    81→    } catch (err) {
    82→    82→      lastErr = err;
    83→    83→    }
    84→    84→  }
    85→    85→  throw lastErr;
    86→    86→}

    88→    88→function parseJson<T>(raw: string): T {
    89→    89→  let text = raw.trim();
    90→    90→  text = text.replace(/^```(?:json)?\s*/i, "").replace(/\s*```$/, "");
    91→    91→  const start = text.search(/[{[]/);
    92→    92→  if (start > 0) text = text.slice(start);
    93→    93→  try {
    94→    94→    return JSON.parse(text) as T;
    95→    95→  } catch {
    96→    96→    /* fall through */
    97→    97→  }
    98→    98→  const lastBrace = Math.max(text.lastIndexOf("}"), text.lastIndexOf("]"));
    99→    99→  if (lastBrace > 0) {
   100→   100→    try {
   101→   101→      return JSON.parse(text.slice(0, lastBrace + 1)) as T;
   102→   102→    } catch {
   103→   103→      /* fall through to structural repair */
   104→   104→    }
   105→   105→  }
   106→   106→  // Production hardening: long LLM outputs can be truncated mid-element.
   107→   107→  // Rebuild a valid prefix: cut at the last structural boundary outside
   108→   108→  // strings, strip dangling commas, then close all open brackets/strings.
   109→   109→  const repaired = repairTruncatedJson(text);
   110→   110→  if (repaired !== null) {
   111→   111→    try {
   112→   112→      return JSON.parse(repaired) as T;
   113→   113→    } catch {
   114→   114→      /* fall through */
   115→   115→    }
   116→   116→  }
   117→   117→  throw new Error("Agent returned unparseable output");
   118→   118→}

   120→   120→function repairTruncatedJson(text: string): string | null {
   121→   121→  // find last structural boundary outside strings
   122→   122→  let inStr = false;
   123→   123→  let esc = false;
   124→   124→  let lastSafe = -1;
   125→   125→  for (let i = 0; i < text.length; i++) {
   126→   126→    const ch = text[i];
   127→   127→    if (inStr) {
   128→   128→      if (esc) esc = false;
   129→   129→      else if (ch === "\\") esc = true;
   130→   130→      else if (ch === '"') inStr = false;
   131→   131→      continue;
   132→   132→    }
   133→   133→    if (ch === '"') {
   134→   134→      inStr = true;
   135→   135→      continue;
   136→   136→    }
   137→   137→    if (ch === "," || ch === "}" || ch === "]") lastSafe = i;
   138→   138→  }
   139→   139→  if (lastSafe < 0) return null;
   140→   140→  let cut = text.slice(0, lastSafe + 1);
   141→   141→  // re-scan the cut for unclosed structures
   142→   142→  const stack: string[] = [];
   143→   143→  let s = false;
   144→   144→  let e = false;
   145→   145→  for (const ch of cut) {
   146→   146→    if (s) {
   147→   147→      if (e) e = false;
   148→   148→      else if (ch === "\\") e = true;
   149→   149→      else if (ch === '"') s = false;
   150→   150→      continue;
   151→   151→    }
   152→   152→    if (ch === '"') {
   153→   153→      s = true;
   154→   154→      continue;
   155→   155→    }
   156→   156→    if (ch === "{") stack.push("}");
   157→   157→    else if (ch === "[") stack.push("]");
   158→   158→    else if (ch === "}" || ch === "]") stack.pop();
   159→   159→  }
   160→   160→  if (s) cut += '"';
   161→   161→  cut = cut.replace(/,\s*$/, "");
   162→   162→  while (stack.length) cut += stack.pop();
   163→   163→  return cut;
   164→   164→}

   186→   186→function evidenceBlock(evidences: { code: string; title: string; sourceType: string; language: string; content: string }[]): string {
   187→   187→  if (!evidences.length) return "(no evidence sources provided yet)";
   188→   188→  // Audit fix: a hard 1200-char per-evidence window was silently blinding
   189→   189→  // every agent to everything beyond it (confirmed by the truncation probe).
   190→   190→  // Budget: up to 4000 chars per evidence, 16000 chars total; when the total
   191→   191→  // overflows, shrink the per-evidence window fairly but never below 1200.
   192→   192→  const TOTAL_BUDGET = 16000;
   193→   193→  const PER_MAX = 4000;
   194→   194→  const total = evidences.reduce((s, e) => s + e.content.length, 0);
   195→   195→  let per = PER_MAX;
   196→   196→  if (total > TOTAL_BUDGET) {
   197→   197→    per = Math.max(1200, Math.floor(TOTAL_BUDGET / evidences.length));
   198→   198→  }
   199→   199→  return evidences
   200→   200→    .map((e) => {
   201→   201→      const body =
   202→   202→        e.content.length > per
   203→   203→          ? `${e.content.slice(0, per)}\n[…truncated for this stage — ${e.content.length - per} more characters exist in this source]`
   204→   204→          : e.content;
   205→   205→      return `[${e.code}] (${e.sourceType}, language: ${e.language}) "${e.title}"\n${body}`;
   206→   206→    })
   207→   207→    .join("\n\n");
   208→   208→}

   824→   824→    const project = await db.project.findUniqueOrThrow({
   825→   825→      where: { id: projectId },
   826→   826→      include: { reqs: true, questions: true, candidates: true, evidences: true },
   827→   827→    });
   828→   828→    const usage = newUsage();
   829→   829→
   830→   830→    const prompt = `A requirements project just completed. Harvest reusable knowledge for FUTURE projects on this platform.
   831→   831→
   832→   832→PROJECT: ${project.name}
   833→   833→BRIEF: ${briefBlock(project.brief)}
   834→   834→DOMAIN EVIDENCE: ${evidenceBlock(project.evidences).slice(0, 2000)}
   835→   835→SELECTED PROBLEM: ${project.candidates.find((c) => c.selected)?.title ?? "(none)"}
   836→   836→Q&A OUTCOMES:
   837→   837→${project.questions.filter((q) => q.answer).map((q) => `- Q: ${q.text.slice(0, 100)} → A: ${(q.answer ?? "").slice(0, 100)}`).join("\n") || "(none)"}
   838→   838→
   839→   839→TASK: Extract (a) 3-5 domain GLOSSARY terms with crisp definitions, and (b) 2-4 reusable LESSONS — what a future analyst should do differently on a similar project (question-outcome lessons, elicitation tactics, domain pitfalls). Lessons must be general enough to carry across projects.
   840→   840→
   841→   841→Return JSON: {"glossary": [{"term": "...", "definition": "..."}], "lessons": [{"term": "short lesson title", "detail": "..."}]}`;
   842→   842→
   843→   843→    const out = await callLLMJson<{
   844→   844→      glossary: { term: string; definition: string }[];
   845→   845→      lessons: { term: string; detail: string }[];
   846→   846→    }>(prompt, usage);
   847→   847→
   848→   848→    for (const g of (out.glossary ?? []).slice(0, 5)) {
   849→   849→      if (!g.term) continue;
   850→   850→      await db.memoryEntry.create({
   851→   851→        data: { projectId, kind: "glossary", term: String(g.term).slice(0, 120), detail: String(g.definition ?? "").slice(0, 400) },
   852→   852→      });
   853→   853→    }
   854→   854→    for (const l of (out.lessons ?? []).slice(0, 4)) {
   855→   855→      if (!l.term) continue;
   856→   856→      await db.memoryEntry.create({
   857→   857→        data: { projectId: null, kind: "lesson", term: String(l.term).slice(0, 120), detail: String(l.detail ?? "").slice(0, 400) },
   858→   858→      });
   859→   859→    }
   860→   860→  } catch {
   861→   861→    // memory harvest is best-effort
   862→   862→  }
   863→   863→}
